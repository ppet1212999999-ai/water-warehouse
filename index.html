function doPost(e){ var data = e.postData.contents; var props = PropertiesService.getScriptProperties(); // 支援同步碼 var code = "default"; try{ if(e.queryString) { var m=e.queryString.match(/code=([^&]+)/); if(m) code=decodeURIComponent(m[1]); } }catch(err){} props.setProperty("WATER_WAREHOUSE_"+code, data); props.setProperty("WATER_WAREHOUSE", data); // 相容舊版 return ContentService.createTextOutput(JSON.stringify({ok:true, len:data.length})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e){ var props = PropertiesService.getScriptProperties(); var code = "default"; try{ if(e && e.parameter && e.parameter.code) code=e.parameter.code; }catch(err){} var data = props.getProperty("WATER_WAREHOUSE_"+code) || props.getProperty("WATER_WAREHOUSE") || "[]"; return ContentService.createTextOutput(data).setMimeType(ContentService.MimeType.JSON); }
